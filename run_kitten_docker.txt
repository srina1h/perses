# Build and run commands for kitten fuzzing with manual V8 instrumentation

# Build the instrumented V8 Docker image:
sudo docker buildx build --builder isolated-builder --load -f Dockerfile.instrumented --build-arg CACHE_BUST=$(date +%s) -t kitten_v8_instrumented .

# Run the container:
sudo docker run --privileged -v $(pwd)/results:/workspace/kitten/findings_v8 kitten_v8_instrumented
