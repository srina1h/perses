# Build and run commands for kitten fuzzing with V8 (no allowlist guidance for now)

# Build the V8 Docker image:
sudo docker buildx build --builder isolated-builder --load -f Dockerfile.instrumented --build-arg CACHE_BUST=$(date +%s) --build-arg V8_BUILD_VERSION=6 -t kitten_v8_instrumented .

# Run the container:
sudo docker run --privileged -v $(pwd)/results:/workspace/kitten/findings_v8 kitten_v8_instrumented
